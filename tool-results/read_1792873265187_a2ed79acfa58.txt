    70→    70→function ServiceFormDialog({
    71→    71→  open,
    72→    72→  onOpenChange,
    73→    73→  editing,
    74→    74→}: {
    75→    75→  open: boolean;
    76→    76→  onOpenChange: (v: boolean) => void;
    77→    77→  editing: Service | null;
    78→    78→}) {
    79→    79→  const { addService, updateService } = useStudioStore();
    80→    80→  const [name, setName] = useState(editing?.name ?? "");
    81→    81→  const [description, setDescription] = useState(editing?.description ?? "");
    82→    82→  const [duration, setDuration] = useState(String(editing?.durationMin ?? 60));
    83→    83→  const [price, setPrice] = useState(String(editing?.price ?? ""));
    84→    84→  const [saving, setSaving] = useState(false);
    85→    85→  const [key, setKey] = useState(0);
    86→    86→
    87→    87→  // reset on open change
    88→    88→  if (open && key !== 1) {
    89→    89→    setName(editing?.name ?? "");
    90→    90→    setDescription(editing?.description ?? "");
    91→    91→    setDuration(String(editing?.durationMin ?? 60));
    92→    92→    setPrice(String(editing?.price ?? ""));
    93→    93→    setKey(1);
    94→    94→  }
    95→    95→  if (!open && key !== 0) setKey(0);
    96→    96→
    97→    97→  const save = async () => {
    98→    98→    if (!name.trim() || !price) {
    99→    99→      toast({ title: "Preencha nome e preço.", variant: "destructive" });
   100→   100→      return;
   101→   101→    }
   102→   102→    const data = {
   103→   103→      name: name.trim(),
   104→   104→      description: description.trim(),
   105→   105→      durationMin: Number(duration),
   106→   106→      price: Number(price),
   107→   107→      image: editing?.image ?? "/images/gallery1.png",
   108→   108→      active: editing?.active ?? true,
   109→   109→    };
   110→   110→    setSaving(true);
   111→   111→    try {
   112→   112→      if (editing) {
   113→   113→        await updateService(editing.id, data);
   114→   114→        toast({ title: "Serviço atualizado!" });
   115→   115→      } else {
   116→   116→        await addService(data);
   117→   117→        toast({ title: "Serviço criado!" });
   118→   118→      }
   119→   119→      onOpenChange(false);
   120→   120→    } catch (e) {
   121→   121→      toast({ title: "Erro ao salvar", description: e instanceof Error ? e.message : "Tente novamente.", variant: "destructive" });
   122→   122→    } finally {
   123→   123→      setSaving(false);
   124→   124→    }
   125→   125→  };
   126→   126→
   127→   127→  return (
   128→   128→    <Dialog open={open} onOpenChange={onOpenChange}>
   129→   129→      <DialogContent className="max-w-md">
   130→   130→        <DialogHeader>
   131→   131→          <DialogTitle className="font-display">{editing ? "Editar serviço" : "Novo serviço"}</DialogTitle>
   132→   132→        </DialogHeader>
   133→   133→        <div className="space-y-3">
   134→   134→          <div className="space-y-1.5">
   135→   135→            <Label>Nome</Label>
   136→   136→            <Input value={name} onChange={(e) => setName(e.target.value)} placeholder="Ex: Manicure Clássica" />
   137→   137→          </div>
   138→   138→          <div className="space-y-1.5">
   139→   139→            <Label>Descrição</Label>
   140→   140→            <Input value={description} onChange={(e) => setDescription(e.target.value)} placeholder="Descrição curta" />
   141→   141→          </div>
   142→   142→          <div className="grid grid-cols-2 gap-3">
   143→   143→            <div className="space-y-1.5">
   144→   144→              <Label>Duração (min)</Label>
   145→   145→              <Select value={duration} onValueChange={setDuration}>
   146→   146→                <SelectTrigger><SelectValue /></SelectTrigger>
   147→   147→                <SelectContent className="max-h-60 overflow-y-auto custom-scrollbar">
   148→   148→                  {[30, 45, 60, 75, 90, 120, 150, 180].map((m) => (
   149→   149→                    <SelectItem key={m} value={String(m)}>{m} min</SelectItem>
   150→   150→                  ))}
   151→   151→                </SelectContent>
   152→   152→              </Select>
   153→   153→            </div>
   154→   154→            <div className="space-y-1.5">
   155→   155→              <Label>Preço (R$)</Label>
   156→   156→              <Input type="number" min={0} value={price} onChange={(e) => setPrice(e.target.value)} placeholder="45" />
   157→   157→            </div>
   158→   158→          </div>
   159→   159→        </div>
   160→   160→        <DialogFooter>
   161→   161→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancelar</Button>
   162→   162→          <Button onClick={save} disabled={saving} className="btn-gold-shimmer text-white font-semibold">{saving ? "Salvando…" : "Salvar"}</Button>
   163→   163→        </DialogFooter>
   164→   164→      </DialogContent>
   165→   165→    </Dialog>
   166→   166→  );
   167→   167→}
