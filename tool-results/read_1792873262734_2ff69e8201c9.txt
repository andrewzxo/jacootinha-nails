    70→function ServiceFormDialog({
    71→  open,
    72→  onOpenChange,
    73→  editing,
    74→}: {
    75→  open: boolean;
    76→  onOpenChange: (v: boolean) => void;
    77→  editing: Service | null;
    78→}) {
    79→  const { addService, updateService } = useStudioStore();
    80→  const [name, setName] = useState(editing?.name ?? "");
    81→  const [description, setDescription] = useState(editing?.description ?? "");
    82→  const [duration, setDuration] = useState(String(editing?.durationMin ?? 60));
    83→  const [price, setPrice] = useState(String(editing?.price ?? ""));
    84→  const [saving, setSaving] = useState(false);
    85→  const [key, setKey] = useState(0);
    86→
    87→  // reset on open change
    88→  if (open && key !== 1) {
    89→    setName(editing?.name ?? "");
    90→    setDescription(editing?.description ?? "");
    91→    setDuration(String(editing?.durationMin ?? 60));
    92→    setPrice(String(editing?.price ?? ""));
    93→    setKey(1);
    94→  }
    95→  if (!open && key !== 0) setKey(0);
    96→
    97→  const save = async () => {
    98→    if (!name.trim() || !price) {
    99→      toast({ title: "Preencha nome e preço.", variant: "destructive" });
   100→      return;
   101→    }
   102→    const data = {
   103→      name: name.trim(),
   104→      description: description.trim(),
   105→      durationMin: Number(duration),
   106→      price: Number(price),
   107→      image: editing?.image ?? "/images/gallery1.png",
   108→      active: editing?.active ?? true,
   109→    };
   110→    setSaving(true);
   111→    try {
   112→      if (editing) {
   113→        await updateService(editing.id, data);
   114→        toast({ title: "Serviço atualizado!" });
   115→      } else {
   116→        await addService(data);
   117→        toast({ title: "Serviço criado!" });
   118→      }
   119→      onOpenChange(false);
   120→    } catch (e) {
   121→      toast({ title: "Erro ao salvar", description: e instanceof Error ? e.message : "Tente novamente.", variant: "destructive" });
   122→    } finally {
   123→      setSaving(false);
   124→    }
   125→  };
   126→
   127→  return (
   128→    <Dialog open={open} onOpenChange={onOpenChange}>
   129→      <DialogContent className="max-w-md">
   130→        <DialogHeader>
   131→          <DialogTitle className="font-display">{editing ? "Editar serviço" : "Novo serviço"}</DialogTitle>
   132→        </DialogHeader>
   133→        <div className="space-y-3">
   134→          <div className="space-y-1.5">
   135→            <Label>Nome</Label>
   136→            <Input value={name} onChange={(e) => setName(e.target.value)} placeholder="Ex: Manicure Clássica" />
   137→          </div>
   138→          <div className="space-y-1.5">
   139→            <Label>Descrição</Label>
   140→            <Input value={description} onChange={(e) => setDescription(e.target.value)} placeholder="Descrição curta" />
   141→          </div>
   142→          <div className="grid grid-cols-2 gap-3">
   143→            <div className="space-y-1.5">
   144→              <Label>Duração (min)</Label>
   145→              <Select value={duration} onValueChange={setDuration}>
   146→                <SelectTrigger><SelectValue /></SelectTrigger>
   147→                <SelectContent className="max-h-60 overflow-y-auto custom-scrollbar">
   148→                  {[30, 45, 60, 75, 90, 120, 150, 180].map((m) => (
   149→                    <SelectItem key={m} value={String(m)}>{m} min</SelectItem>
   150→                  ))}
   151→                </SelectContent>
   152→              </Select>
   153→            </div>
   154→            <div className="space-y-1.5">
   155→              <Label>Preço (R$)</Label>
   156→              <Input type="number" min={0} value={price} onChange={(e) => setPrice(e.target.value)} placeholder="45" />
   157→            </div>
   158→          </div>
   159→        </div>
   160→        <DialogFooter>
   161→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancelar</Button>
   162→          <Button onClick={save} disabled={saving} className="btn-gold-shimmer text-white font-semibold">{saving ? "Salvando…" : "Salvar"}</Button>
   163→        </DialogFooter>
   164→      </DialogContent>
   165→    </Dialog>
   166→  );
   167→}
